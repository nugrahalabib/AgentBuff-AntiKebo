import { execFileSync } from "node:child_process";
import { createHash, randomUUID } from "node:crypto";
import { and, eq, inArray } from "drizzle-orm";
import { encode } from "next-auth/jwt";
import { chromium, type Browser } from "playwright";
import { db } from "@/lib/db";
import * as schema from "@/lib/db/schema";
import { issuerMasuk } from "@/lib/masuk/alamat";
import { pastikanGrant } from "@/lib/masuk/grant";
import { tandatanganiJwt } from "@/lib/masuk/jwt";
import { kunciMasuk } from "@/lib/masuk/kunci";
import { penilai } from "./_bym-uji";
import { headerCookieUntuk, NAMA_COOKIE, pastikanSesiDiterima } from "./_sesi-tempaan";

const { nilai, hasil } = penilai();
const BASIS = "https://agentbuff.id";
const TUYA = "https://tuya.agentbuff.id";
const KUNCI = "tuya-mcp";
const HARI = 86_400_000;

const aman = (s: string) => s.replace(/'/g, "''");
function psqlTuya(sql: string): string {
  return execFileSync("docker", ["exec", "tuya-db", "psql", "-U", "tuya_super", "-d", "tuya", "-tAc", sql], { encoding: "utf8", timeout: 30_000 }).trim();
}

async function api(kuki: string, metode: string, jalur: string, badan?: unknown) {
  const r = await fetch(`${BASIS}${jalur}`, { method: metode, headers: { cookie: kuki, "content-type": "application/json" }, body: badan === undefined ? undefined : JSON.stringify(badan) });
  return { status: r.status, isi: ((await r.json().catch(() => ({}))) ?? {}) as Record<string, unknown> };
}

async function webhookLunas(orderId: string, grossRp: number): Promise<number> {
  const gross = `${grossRp}.00`;
  const signature = createHash("sha512").update(orderId + "200" + gross + process.env.MIDTRANS_SERVER_KEY!).digest("hex");
  const r = await fetch(`${BASIS}/api/billing/webhook`, {
    method: "POST",
    headers: { "content-type": "application/json" },
    body: JSON.stringify({ order_id: orderId, status_code: "200", gross_amount: gross, signature_key: signature, transaction_status: "settlement", transaction_id: `uji-${randomUUID()}`, fraud_status: "accept", payment_type: "qris" }),
  });
  return r.status;
}

async function mintaTokenTuya(uid: string, email: string): Promise<{ status: number; token: string }> {
  const g = await pastikanGrant(uid, "tuya", { hidupkanLagi: false });
  const iat = Math.floor(Date.now() / 1000);
  const as = tandatanganiJwt({ iss: issuerMasuk(), aud: "tuya", sub: g.sub, email, email_verified: true, purpose: "mcp_token", iat, exp: iat + 120, jti: randomUUID() }, kunciMasuk(), "mcp-token+jwt");
  const r = await fetch(`${TUYA}/api/agentbuff/mcp-token`, { method: "POST", headers: { authorization: `Bearer ${as}`, "content-type": "application/json" }, body: "{}" });
  const b = (await r.json().catch(() => ({}))) as { token?: string };
  return { status: r.status, token: String(b.token ?? "") };
}

let urut = 0;
async function rpc(token: string, method: string, params: Record<string, unknown> = {}) {
  const r = await fetch(`${TUYA}/mcp`, {
    method: "POST",
    headers: { "content-type": "application/json", accept: "application/json, text/event-stream", "mcp-protocol-version": "2025-06-18", authorization: `Bearer ${token}` },
    body: JSON.stringify({ jsonrpc: "2.0", id: ++urut, method, params }),
  });
  const teks = await r.text();
  let json: { result?: Record<string, unknown>; error?: { code: number; message?: string } } | null = null;
  try {
    json = JSON.parse(teks);
  } catch {
    for (const l of teks.split("\n").filter((x) => x.startsWith("data:"))) {
      try {
        json = JSON.parse(l.slice(5).trim());
      } catch {
        /* abaikan */
      }
    }
  }
  return { http: r.status, result: json?.result ?? {}, error: json?.error ?? null };
}
async function alat(token: string, nama: string, argumen: Record<string, unknown> = {}) {
  const r = await rpc(token, "tools/call", { name: nama, arguments: argumen });
  return { sc: (r.result.structuredContent ?? {}) as Record<string, unknown>, galat: r.result.isError === true, rpc: r.error };
}

async function ent(userId: string) {
  const [e] = await db.select().from(schema.skillEntitlements).where(and(eq(schema.skillEntitlements.userId, userId), eq(schema.skillEntitlements.catalogKey, KUNCI))).limit(1);
  return e;
}

async function konteks(browser: Browser, token: string) {
  const ctx = await browser.newContext({ locale: "id-ID", viewport: { width: 390, height: 844 } });
  await ctx.addCookies([{ name: NAMA_COOKIE, value: token, domain: "agentbuff.id", path: "/", secure: true, httpOnly: true, sameSite: "Lax" }]);
  return ctx;
}

async function main() {
  if (process.env.MIDTRANS_IS_PRODUCTION === "true") {
    console.error("Midtrans mode PRODUKSI - bukti ini hanya boleh jalan di sandbox.");
    process.exit(2);
  }
  const id = randomUUID();
  const email = `tuya-beli-${id.slice(0, 8)}@uji.internal`;
  await db.insert(schema.users).values({ id, email, name: "Uji Beli Tuya" });
  await db.insert(schema.userTrials).values({ userId: id, endsAt: new Date(Date.now() + 30 * HARI), status: "active" });
  const kuki = await headerCookieUntuk(id);
  const tokenSesi = await encode({ token: { id, sub: id, email, name: "Uji Beli Tuya" }, secret: process.env.AUTH_SECRET!, salt: NAMA_COOKIE });
  const browser = await chromium.launch();
  try {
    await pastikanSesiDiterima(BASIS, kuki);

    // ---------------------------------------------------------------- A
    console.log("\nA. Dijual di Marketplace");
    const katalog = (await (await fetch(`${BASIS}/api/skills/buffhub?locale=id`)).json()) as { skills: Array<Record<string, unknown>> };
    const p0 = katalog.skills.find((s) => s.slug === KUNCI);
    nilai("Tuya MCP ada di katalog, status available", p0?.status === "available", JSON.stringify({ status: p0?.status }));
    nilai("sekali bayar Rp29.000", p0?.billing === "one_time" && p0?.priceRp === 29_000, JSON.stringify({ billing: p0?.billing, harga: p0?.priceRp }));
    const ctxA = await browser.newContext({ locale: "id-ID", viewport: { width: 390, height: 844 } });
    const pa = await ctxA.newPage();
    await pa.goto(`${BASIS}/marketplace?produk=${KUNCI}`, { waitUntil: "domcontentloaded" });
    await pa.getByText("Tuya MCP").first().waitFor({ timeout: 30_000 });
    await pa.waitForTimeout(2500);
    const teksA = await pa.getByRole("dialog").first().innerText();
    nilai("detail publik: tanpa 'Segera hadir'", !/segera hadir|coming soon/i.test(teksA));
    nilai("detail publik: Rp29.000 tampil", /29\.000/.test(teksA));
    nilai("detail publik: tutorial 'tempel di chat agenmu'", /chat agenmu/i.test(teksA));
    await ctxA.close();

    // ---------------------------------------------------------------- B
    console.log("\nB. Sebelum membeli: masuk ditolak sopan");
    const ctxB = await konteks(browser, tokenSesi);
    const pb = await ctxB.newPage();
    await pb.goto(`${TUYA}/masuk`, { waitUntil: "domcontentloaded" });
    await pb.getByRole("link", { name: /Masuk dengan AgentBuff/i }).first().click();
    await pb.waitForURL((u) => !u.href.includes("/auth/agentbuff/start"), { timeout: 30_000 }).catch(() => {});
    await pb.waitForTimeout(3000);
    const urlB = pb.url();
    nilai("tombol masuk benar-benar pindah ke AgentBuff (bukan macet)", !urlB.includes("_rsc=") && (urlB.includes("agentbuff.id/masuk") || urlB.includes(`${TUYA}/masuk?`)), urlB);
    nilai("belum membeli -> TIDAK sampai ke /app", !/tuya\.agentbuff\.id\/app/.test(urlB), urlB);
    await ctxB.close();

    // ---------------------------------------------------------------- C
    console.log("\nC. Beli & bayar");
    const r = await api(kuki, "POST", "/api/billing/skill", { skillKey: KUNCI, bulan: 1, expectedPriceRp: 29_000 });
    nilai("pesanan dibuat dengan token Snap", r.status === 200 && typeof r.isi.token === "string", JSON.stringify(r.isi).slice(0, 200));
    const [t] = r.isi.transactionId ? await db.select().from(schema.transactions).where(eq(schema.transactions.id, String(r.isi.transactionId))).limit(1) : [];
    nilai("ditagih Rp29.000", t?.amountRp === 29_000, String(t?.amountRp));
    const w = t?.midtransOrderId ? await webhookLunas(t.midtransOrderId, 29_000) : 0;
    const e1 = await ent(id);
    nilai("webhook Midtrans diterima -> kepemilikan aktif", w === 200 && e1?.status === "active", `${w} ${e1?.status}`);
    nilai("sekali bayar: tanpa tanggal berakhir", e1?.billing === "one_time" && !e1.expiresAt, `${e1?.billing} ${e1?.expiresAt?.toISOString()}`);

    // ---------------------------------------------------------------- D
    console.log("\nD. Masuk dengan AgentBuff");
    const ctxD = await konteks(browser, tokenSesi);
    const pd = await ctxD.newPage();
    await pd.goto(`${TUYA}/masuk`, { waitUntil: "domcontentloaded" });
    await pd.getByRole("link", { name: /Masuk dengan AgentBuff/i }).first().click();
    await pd.waitForURL(/agentbuff\.id\/masuk\/authorize|tuya\.agentbuff\.id\/app/, { timeout: 30_000 }).catch(() => {});
    const kartu = pd.locator('[data-masuk="konfirmasi"]');
    if (await kartu.isVisible({ timeout: 8000 }).catch(() => false)) {
      nilai("layar konfirmasi menyebut email pemilik", (await pd.locator("[data-masuk-email]").innerText()).includes(email));
      await kartu.locator('button[name="keputusan"][value="izinkan"]').click();
    }
    await pd.waitForURL(/tuya\.agentbuff\.id\/app/, { timeout: 45_000 }).catch(() => {});
    nilai("kembali ke tuya.agentbuff.id SUDAH MASUK", /tuya\.agentbuff\.id\/app/.test(pd.url()), pd.url());
    nilai("rumah belum tersambung -> langsung ke layar Sambungkan", /\/app\/sambungkan/.test(pd.url()), pd.url());
    await pd.waitForTimeout(1500);
    nilai("layar Sambungkan tampil (langkah 1 dari 3, boleh tempel di chat)", /Sambungkan rumahmu|Connect your home/i.test(await pd.locator("body").innerText()) && /chat/i.test(await pd.locator("body").innerText()));
    nilai("akun pemilik tercatat di Tuya MCP", psqlTuya(`select count(*) from pengguna where email='${aman(email)}';`) === "1");
    await ctxD.close();

    // ---------------------------------------------------------------- E
    console.log("\nE. Alat MCP lewat token otomatis");
    const m = await mintaTokenTuya(id, email);
    nilai("token MCP otomatis terbit (tuya_...)", m.status === 200 && /^tuya_/.test(m.token), String(m.status));
    const daftar = await rpc(m.token, "tools/list");
    const nama = ((daftar.result.tools ?? []) as Array<{ name: string }>).map((x) => x.name);
    const wajib = ["connect_home", "control_device", "control_devices", "save_scene", "run_scene", "create_schedule", "create_automation", "list_automations", "capture_camera", "notify_me", "get_energy_usage", "set_device_preferences"];
    nilai(`daftar alat lengkap (${nama.length} alat)`, wajib.every((x) => nama.includes(x)), wajib.filter((x) => !nama.includes(x)).join(","));
    const st = await alat(m.token, "get_setup_status");
    nilai("get_setup_status: belum tersambung + langkah tuya.ai", st.sc.connected === false && !st.galat, JSON.stringify(st.sc).slice(0, 160));
    const ld = await alat(m.token, "list_devices");
    nilai("alat rumah sebelum tersambung -> not_connected (bukan galat aneh)", ld.galat && ld.sc.error_code === "not_connected", JSON.stringify(ld.sc).slice(0, 160));
    const ch = await alat(m.token, "connect_home", { key: "sk-AZujiPalsuBukanKunci0000000000" });
    nilai("connect_home: kunci palsu ditolak oleh server Tuya sungguhan", ch.galat && ch.sc.error_code === "invalid_key", JSON.stringify(ch.sc).slice(0, 200));
    nilai("kunci palsu TIDAK tersimpan", psqlTuya(`select count(*) from sambungan_tuya s join pengguna p on p.id=s.pengguna_id where p.email='${aman(email)}';`) === "0");
  } finally {
    await browser.close();
    try {
      const u = `(select id from pengguna where email='${aman(email)}')`;
      psqlTuya(
        `begin; delete from otomasi where pengguna_id=${u}; delete from jadwal where pengguna_id=${u}; delete from suasana where pengguna_id=${u}; delete from perangkat where pengguna_id=${u}; delete from sambungan_tuya where pengguna_id=${u}; delete from aktivitas where pengguna_id=${u}; delete from token_mcp where pengguna_id=${u}; delete from sesi where pengguna_id=${u}; delete from status_hak where pengguna_id=${u}; delete from pengguna where email='${aman(email)}'; commit;`,
      );
      console.log(`\n  bersih-bersih Tuya: sisa ${psqlTuya(`select count(*) from pengguna where email='${aman(email)}';`)} akun`);
    } catch (e) {
      console.log("  ⚠ bersih-bersih Tuya gagal:", e instanceof Error ? e.message : e);
    }
    await new Promise((res) => setTimeout(res, 8000)); // struk & email dikirim tanpa ditunggu
    await db.delete(schema.masukGrant).where(eq(schema.masukGrant.userId, id));
    await db.delete(schema.mcpOtomatis).where(eq(schema.mcpOtomatis.userId, id));
    await db.delete(schema.skillEntitlements).where(eq(schema.skillEntitlements.userId, id));
    await db.delete(schema.analyticsEvents).where(eq(schema.analyticsEvents.userId, id));
    await db.delete(schema.emailLog).where(eq(schema.emailLog.userId, id));
    await db.delete(schema.notifications).where(eq(schema.notifications.userId, id));
    await db.delete(schema.containerSkills).where(eq(schema.containerSkills.userId, id));
    await db.delete(schema.transactions).where(eq(schema.transactions.userId, id));
    await db.delete(schema.userTrials).where(eq(schema.userTrials.userId, id));
    await db.delete(schema.users).where(inArray(schema.users.id, [id]));
  }
  const { lulus, gagal } = hasil();
  console.log(`\n${gagal === 0 ? "HIJAU" : "MERAH"} ${lulus}/${lulus + gagal}`);
  process.exit(gagal === 0 ? 0 : 1);
}

main().catch((e) => {
  console.error(e);
  process.exit(1);
});
