async function panggil(kuki: string, metode: string, jalur: string, badan?: unknown): Promise<unknown> {
  const r = await fetch(`${BASIS}${jalur}`, {
    method: metode,
    headers: { cookie: kuki, "content-type": "application/json" },
    body: badan === undefined ? undefined : JSON.stringify(badan),
  });
  const isi = await r.json().catch(() => null);
  if (!r.ok) throw new Error(`${metode} ${jalur} -> ${r.status} ${JSON.stringify(isi)}`);
  return isi;
}

async function main() {
  const kuki = await headerCookie();
  await pastikanSesiDiterima(BASIS, kuki);

  const [ada] = await db.select({ key: schema.skillCatalog.key, status: schema.skillCatalog.status }).from(schema.skillCatalog).where(eq(schema.skillCatalog.key, KUNCI)).limit(1);
  if (!ada) {
    await panggil(kuki, "POST", "/api/admin/catalog", { key: KUNCI, ...PRODUK });
    console.log(`✓ katalog ${KUNCI} dibuat - Rp29.000 sekali bayar, status ${PRODUK.status}`);
  } else {
    const { status: _s, ...tanpaStatus } = PRODUK;
    void _s;
    const naikSaja = ada.status === "available" || !TERBITKAN;
    await panggil(kuki, "PATCH", `/api/admin/catalog/${KUNCI}`, naikSaja ? tanpaStatus : PRODUK);
    console.log(`✓ katalog ${KUNCI} diperbarui (status ${naikSaja ? `tetap ${ada.status}` : `${ada.status} -> available`})`);
  }

  await panggil(kuki, "POST", `/api/admin/catalog/${KUNCI}/mcp`, {
    name: "tuya",
    transport: "http",
    url: "https://tuya.agentbuff.id/mcp",
    tokenHeader: "Authorization",
    tokenPrefix: "Bearer ",
    tokenLabel: "Token Tuya MCP",
    tutorial: TUTORIAL_MANUAL,
  });
  console.log("✓ konektor MCP tuya -> https://tuya.agentbuff.id/mcp");

  const argSkill = process.argv.indexOf("--skill");
  const kandidat = [
    argSkill > 0 ? process.argv[argSkill + 1] : null,
    process.env.TUYA_SKILL_PATH ?? null,
    "/opt/tuya/app/skill/SKILL.md",
    path.join(process.cwd(), "../AgentBuff-Tuya/skill/SKILL.md"),
  ].filter((p): p is string => !!p);
  const berkas = kandidat.find((p) => existsSync(p));
  if (!berkas) throw new Error(`SKILL.md Tuya tidak ditemukan (dicoba: ${kandidat.join(", ")})`);
  const skill = readFileSync(berkas);
  const hashBaru = createHash("sha256").update(skill).digest("hex");
  const [paketLama] = await db
    .select({ files: schema.skillPackages.files })
    .from(schema.skillPackages)
    .where(and(eq(schema.skillPackages.catalogKey, KUNCI), eq(schema.skillPackages.slug, "tuya-mcp")))
    .orderBy(desc(schema.skillPackages.createdAt))
    .limit(1);
  const hashLama = paketLama?.files?.[0]?.contentB64 ? createHash("sha256").update(Buffer.from(paketLama.files[0].contentB64, "base64")).digest("hex") : null;
  const [katalog] = await db.select({ packageId: schema.skillCatalog.packageId }).from(schema.skillCatalog).where(eq(schema.skillCatalog.key, KUNCI)).limit(1);
  if (hashLama === hashBaru && katalog?.packageId) {
    console.log(`✓ SKILL.md pendamping sudah terbaru (${berkas})`);
  } else {
    await panggil(kuki, "POST", `/api/admin/catalog/${KUNCI}/package`, {
      slug: "tuya-mcp",
      version: `tuya-${hashBaru.slice(0, 8)}`,
      files: [{ path: "SKILL.md", contentB64: skill.toString("base64") }],
    });
    console.log(`✓ SKILL.md pendamping diunggah (${berkas}, tuya-${hashBaru.slice(0, 8)})`);
  }
  process.exit(0);
}

main().catch((e) => {
  console.error(e);
  process.exit(1);
});
